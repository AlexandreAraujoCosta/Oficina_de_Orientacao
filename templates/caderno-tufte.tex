% Template pandoc, tema Tufte: margem larga a direita, itens do relatorio ao
% lado do paragrafo que citam. Compila com xelatex.
%
% E o formato natural para o caderno de conferencia, porque o caderno e texto
% anotado: o paragrafo fica na coluna principal e a anotacao ao lado dele, sem
% empurrar a leitura para o pe da pagina. O risco e colisao quando muitos itens
% caem em paragrafos vizinhos; nesse caso use o tema classico.
\documentclass[11pt,a4paper]{article}

\usepackage{fontspec}
\usepackage{polyglossia}
\setdefaultlanguage{portuguese}

\usepackage{geometry}
\geometry{a4paper, top=2.5cm, bottom=2.5cm, left=2.2cm, right=6.8cm,
          marginparwidth=5.4cm, marginparsep=0.6cm}

\usepackage{xcolor}
\definecolor{unbAzul}{HTML}{003366}
\definecolor{unbVerde}{HTML}{006633}
\definecolor{margemCor}{HTML}{55524D}
\definecolor{realce}{HTML}{33477E}

\usepackage{marginnote}
\renewcommand*{\marginfont}{\footnotesize\color{margemCor}}

\usepackage{titlesec}
% Titulo nao se hifeniza: pedido de 31/08/2026. \raggedright desliga a
% justificacao na linha do titulo, e a penalidade proibitiva impede a quebra
% de palavra dentro dele; sem os dois, o LaTeX parte a palavra para justificar.
\newcommand{\semHifen}{\raggedright\hyphenpenalty=10000\exhyphenpenalty=10000}
\titleformat{\section}{\semHifen\color{unbAzul}\bfseries\huge}{}{0pt}{}
\titleformat{\subsection}{\semHifen\color{unbAzul}\bfseries\LARGE}{}{0pt}{}
\titleformat{\subsubsection}{\semHifen\color{unbAzul!85!black}\bfseries\large}{}{0pt}{}

\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\footnotesize\color{margemCor}$if(title)$$title$$endif$}
\fancyfoot[R]{\footnotesize\color{margemCor}\thepage}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\headrule}{\vskip2pt{\color{unbVerde}\hrule height 0.8pt}}

\usepackage{longtable,booktabs,array}
% calc define o \real das larguras de coluna que o pandoc emite para
% tabela de markdown. Sem ele o xelatex para em Missing number no
% primeiro \begin{longtable}. So apareceu em 23/08, no primeiro
% relatorio com tabela no corpo.
\usepackage{calc}
\usepackage{graphicx}
\usepackage{hyperref}
\hypersetup{colorlinks=true, linkcolor=unbAzul, urlcolor=unbAzul,
            pdftitle={$if(title)$$title$$endif$}}

\setlength{\parindent}{0pt}
\setlength{\parskip}{6pt plus 2pt}
\usepackage{setspace}
\onehalfspacing

% \citado{P20}{2.3, 3.4} poe o localizador e os itens na margem.
\newcommand{\citado}[2]{\marginnote{\textbf{\color{realce}#1}\\ #2}}
\newcommand{\numero}[1]{\marginnote{\textbf{#1}}}

$if(highlighting-macros)$
$highlighting-macros$
$endif$

\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}


% Na primeira pagina o cabecalho corrente nao repete o titulo, que esta logo
% abaixo dele. Pedido de 31/08/2026. O rodape com o numero fica.
\fancypagestyle{aberturaOficina}{%
  \fancyhf{}%
  \fancyfoot[C]{\footnotesize\color{margemCor}\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\headrule}{}%
}

\begin{document}
\thispagestyle{aberturaOficina}

$if(title)$
\begin{center}
{\color{unbAzul}\bfseries\LARGE\semHifen $title$\par}
$if(autoria)$
\vspace{5pt}
{\color{unbAzul}\bfseries\Large\semHifen $autoria$\par}
$endif$
\end{center}
\vspace{6pt}
{\color{unbVerde}\hrule height 1pt}
\vspace{14pt}
$endif$

$body$

\end{document}
